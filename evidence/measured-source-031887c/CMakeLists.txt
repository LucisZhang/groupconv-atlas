cmake_minimum_required(VERSION 3.20)
project(groupconv_atlas VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GC_ENABLE_OPENMP "Build real OpenMP implementation" ON)
option(GC_ENABLE_OPENCL "Build OpenCL course backend" ON)
option(GC_ENABLE_CUDA "Build CUDA backend (requires nvcc)" OFF)
option(GC_SANITIZE "Enable CPU address and undefined behavior sanitizers" OFF)
file(GLOB GC_CPU_SOURCES CONFIGURE_DEPENDS csrc/cpu/*.cpp)
add_library(groupconv SHARED csrc/common/common.cpp ${GC_CPU_SOURCES})
target_include_directories(groupconv PUBLIC csrc/common)
target_compile_options(groupconv PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>)
if(GC_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(groupconv PRIVATE OpenMP::OpenMP_CXX)
endif()
if(GC_ENABLE_OPENCL)
  find_package(OpenCL REQUIRED)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS csrc/opencl/conv.cl)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/csrc/opencl/conv.cl" GC_OPENCL_SOURCE)
  configure_file(csrc/opencl/source.h.in generated/opencl_source.h @ONLY)
  target_sources(groupconv PRIVATE csrc/opencl/opencl.cpp)
  target_include_directories(groupconv PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(groupconv PRIVATE OpenCL::OpenCL)
else()
  target_sources(groupconv PRIVATE csrc/opencl/stub.cpp)
endif()
if(GC_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(groupconv PRIVATE csrc/cuda/groupconv.cu)
  target_link_libraries(groupconv PRIVATE CUDA::cudart)
  set_target_properties(groupconv PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(groupconv PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
else()
  target_sources(groupconv PRIVATE csrc/cuda/stub.cpp)
endif()
if(GC_SANITIZE)
  target_compile_options(groupconv PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(groupconv PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(native_cpu tests/native_cpu.cpp)
target_link_libraries(native_cpu PRIVATE groupconv)
if(GC_SANITIZE)
  target_compile_options(native_cpu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(native_cpu PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME native_cpu COMMAND native_cpu)
if(GC_ENABLE_CUDA)
  add_executable(cuda_microbench csrc/cuda/microbench.cu)
  target_link_libraries(cuda_microbench PRIVATE CUDA::cudart)
  set_target_properties(cuda_microbench PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cuda_microbench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  add_test(NAME cuda_microbench_smoke COMMAND cuda_microbench --smoke)
  add_executable(native_cuda tests/native_cuda.cu)
  target_link_libraries(native_cuda PRIVATE groupconv CUDA::cudart)
  set_target_properties(native_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(native_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>)
  add_test(NAME native_cuda COMMAND native_cuda)
  add_test(NAME native_cuda_contracts COMMAND native_cuda --contracts)
  add_test(NAME native_cuda_stress COMMAND native_cuda --stress)
  add_test(NAME native_cuda_graph COMMAND native_cuda --graph)
endif()
